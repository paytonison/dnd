cmake_minimum_required(VERSION 3.24)
project(DungeoningADragon VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(MSVC)
  add_compile_options(/utf-8)
endif()
option(DND_BUILD_GUI "Build the Qt desktop application" ON)
option(DND_BUILD_TESTS "Build automated tests" ON)

find_package(nlohmann_json 3.11 QUIET)
if(NOT nlohmann_json_FOUND)
  include(FetchContent)
  FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(json)
endif()

add_library(dnd_engine OBJECT src/engine.cpp)
add_library(dnd_editions OBJECT src/editions/registry.cpp src/editions/bx.cpp src/editions/srd55.cpp src/editions/srd55_v2.cpp src/editions/srd55_v2_content.cpp src/editions/srd55_features.cpp src/editions/srd55_lifecycle.cpp src/editions/srd55_history.cpp src/editions/srd55_inventory.cpp src/editions/srd55_companions.cpp src/editions/srd55_class_actions.cpp)
foreach(target dnd_engine dnd_editions)
  target_include_directories(${target} PUBLIC include)
  target_link_libraries(${target} PUBLIC nlohmann_json::nlohmann_json)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
add_library(dnd_rules STATIC $<TARGET_OBJECTS:dnd_engine> $<TARGET_OBJECTS:dnd_editions>)
target_include_directories(dnd_rules PUBLIC include)
target_link_libraries(dnd_rules PUBLIC nlohmann_json::nlohmann_json)
add_library(dnd_content STATIC src/content.cpp src/persistence.cpp)
target_link_libraries(dnd_content PUBLIC dnd_rules)
add_executable(dnd-cli src/cli/main.cpp)
target_link_libraries(dnd-cli PRIVATE dnd_content)
target_compile_definitions(dnd-cli PRIVATE DND_DATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/data/packs")

if(DND_BUILD_GUI)
  find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets PrintSupport Test)
  set(CMAKE_AUTOMOC ON)
  set(CMAKE_AUTORCC ON)
  file(GLOB APP_SOURCES CONFIGURE_DEPENDS src/app/*.cpp src/app/*.hpp)
  list(FILTER APP_SOURCES EXCLUDE REGEX "/main\\.cpp$")
  add_library(dnd_gui STATIC ${APP_SOURCES})
  target_include_directories(dnd_gui PUBLIC src/app)
  target_link_libraries(dnd_gui PUBLIC dnd_content Qt6::Widgets Qt6::PrintSupport)
  target_compile_definitions(dnd_gui PUBLIC DND_DATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/data/packs")
  add_executable(dnd_desktop MACOSX_BUNDLE WIN32 src/app/main.cpp assets/icons.qrc)
  target_link_libraries(dnd_desktop PRIVATE dnd_gui)
  set_target_properties(dnd_desktop PROPERTIES OUTPUT_NAME "Dungeoning a Dragon" MACOSX_BUNDLE_GUI_IDENTIFIER "org.paytonison.dungeoning-a-dragon" MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}" MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}")
  if(APPLE)
    set_source_files_properties(assets/icons/app-icon.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
    set_source_files_properties(assets/icons/ATTRIBUTION.md PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
    target_sources(dnd_desktop PRIVATE assets/icons/app-icon.icns assets/icons/ATTRIBUTION.md)
    set_target_properties(dnd_desktop PROPERTIES MACOSX_BUNDLE_ICON_FILE "app-icon.icns")
    add_custom_command(TARGET dnd_desktop POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/data/packs" "$<TARGET_BUNDLE_CONTENT_DIR:dnd_desktop>/Resources/packs")
  else()
    add_custom_command(TARGET dnd_desktop POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/data/packs" "$<TARGET_FILE_DIR:dnd_desktop>/packs")
  endif()
  if(WIN32)
    enable_language(RC)
    configure_file(assets/app-icon.rc.in ${CMAKE_CURRENT_BINARY_DIR}/app-icon.rc @ONLY)
    target_sources(dnd_desktop PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/app-icon.rc)
  elseif(UNIX AND NOT APPLE)
    install(FILES assets/dungeoning-a-dragon.desktop DESTINATION share/applications)
    install(FILES assets/icons/app-icon.png DESTINATION share/icons/hicolor/1024x1024/apps RENAME dungeoning-a-dragon.png)
  endif()
  install(TARGETS dnd_desktop BUNDLE DESTINATION . RUNTIME DESTINATION bin)
endif()
if(DND_BUILD_TESTS)
  enable_testing()
  find_package(Catch2 3 QUIET)
  if(NOT Catch2_FOUND)
    include(FetchContent)
    FetchContent_Declare(Catch2 URL https://github.com/catchorg/Catch2/archive/refs/tags/v3.8.1.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_MakeAvailable(Catch2)
  endif()
  add_executable(dnd_tests tests/test_core.cpp tests/test_bx.cpp tests/test_srd55.cpp tests/test_srd55_v2_content.cpp tests/test_srd55_features.cpp tests/test_srd55_v2.cpp tests/test_srd55_complete.cpp tests/test_srd55_history.cpp tests/test_srd55_lifecycle.cpp tests/test_srd55_inventory.cpp tests/test_srd55_companions.cpp tests/test_srd55_class_actions.cpp)
  target_link_libraries(dnd_tests PRIVATE dnd_content Catch2::Catch2WithMain)
  target_compile_definitions(dnd_tests PRIVATE DND_DATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/data/packs")
  add_test(NAME rules-and-content COMMAND dnd_tests)
  if(DND_BUILD_GUI)
    add_executable(dnd_gui_tests tests/test_gui.cpp)
    target_link_libraries(dnd_gui_tests PRIVATE dnd_gui Qt6::Test)
    add_test(NAME desktop-workflow COMMAND dnd_gui_tests)
    set_tests_properties(desktop-workflow PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  endif()
endif()
install(TARGETS dnd-cli RUNTIME DESTINATION bin)
install(DIRECTORY data/packs DESTINATION share/dungeoning-a-dragon)
install(FILES LICENSE DESTINATION share/dungeoning-a-dragon)
install(FILES assets/icons/ATTRIBUTION.md DESTINATION share/dungeoning-a-dragon RENAME ICON-ATTRIBUTION.md)
set(CPACK_PACKAGE_NAME "DungeoningADragon")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
include(CPack)
